-- Beri jeda 3 detik agar aman
task.wait(3)

local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer
local PlayerGui = LocalPlayer:WaitForChild("PlayerGui")
local StarterGui = game:GetService("StarterGui")
local UserInputService = game:GetService("UserInputService")
local RunService = game:GetService("RunService")
local Workspace = game:GetService("Workspace")

-- Fungsi Pop-up Peringatan (Warning Prompt)
local function showWarningPrompt()
    local warnGui = Instance.new("ScreenGui")
    warnGui.Name = "NDSWarnGui"
    warnGui.ResetOnSpawn = false
    warnGui.Parent = PlayerGui

    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(0, 280, 0, 160)
    frame.Position = UDim2.new(0.5, -140, 0.5, -80)
    frame.BackgroundColor3 = Color3.fromRGB(20, 20, 30)
    frame.BorderSizePixel = 0
    frame.Parent = warnGui

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 12)
    corner.Parent = frame

    local uiStroke = Instance.new("UIStroke")
    uiStroke.Color = Color3.fromRGB(255, 60, 60)
    uiStroke.Thickness = 2
    uiStroke.Parent = frame

    local title = Instance.new("TextLabel")
    title.Size = UDim2.new(1, 0, 0, 35)
    title.Position = UDim2.new(0, 0, 0, 10)
    title.BackgroundTransparency = 1
    title.Text = "⚠️ WARNING ⚠️"
    title.TextColor3 = Color3.fromRGB(255, 60, 60)
    title.TextSize = 16
    title.Font = Enum.Font.GothamBold
    title.Parent = frame

    local desc = Instance.new("TextLabel")
    desc.Size = UDim2.new(1, -20, 0, 60)
    desc.Position = UDim2.new(0, 10, 0, 45)
    desc.BackgroundTransparency = 1
    desc.Text = "Are you sure? If your account gets banned, it is not our fault and you take full risk. Do you want to continue?"
    desc.TextColor3 = Color3.fromRGB(220, 220, 220)
    desc.TextSize = 12
    desc.Font = Enum.Font.Gotham
    desc.TextWrapped = true
    desc.Parent = frame

    -- Tombol Yes
    local yesBtn = Instance.new("TextButton")
    yesBtn.Size = UDim2.new(0, 110, 0, 32)
    yesBtn.Position = UDim2.new(0.08, 0, 0.75, 0)
    yesBtn.BackgroundColor3 = Color3.fromRGB(0, 150, 80)
    yesBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    yesBtn.TextSize = 13
    yesBtn.Font = Enum.Font.GothamBold
    yesBtn.Text = "Yes, I Agree"
    yesBtn.Parent = frame

    local yesCorner = Instance.new("UICorner")
    yesCorner.CornerRadius = UDim.new(0, 6)
    yesCorner.Parent = yesBtn

    -- Tombol No
    local noBtn = Instance.new("TextButton")
    noBtn.Size = UDim2.new(0, 110, 0, 32)
    noBtn.Position = UDim2.new(0.54, 0, 0.75, 0)
    noBtn.BackgroundColor3 = Color3.fromRGB(180, 50, 50)
    noBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    noBtn.TextSize = 13
    noBtn.Font = Enum.Font.GothamBold
    noBtn.Text = "No (Cancel)"
    noBtn.Parent = frame

    local noCorner = Instance.new("UICorner")
    noCorner.CornerRadius = UDim.new(0, 6)
    noCorner.Parent = noBtn

    -- Logika Tombol
    local agreed = false
    yesBtn.MouseButton1Click:Connect(function()
        agreed = true
        warnGui:Destroy()
    end)

    noBtn.MouseButton1Click:Connect(function()
        warnGui:Destroy()
    end)

    -- Tunggu sampai player memilih
    while warnGui.Parent do
        task.wait(0.1)
    end

    return agreed
end

-- Jalankan prompt peringatan dulu, kalau klik Yes baru buka GUI utama
if showWarningPrompt() then

    -- Hapus GUI lama jika skrip dieksekusi ulang
    if PlayerGui:FindFirstChild("NDSUltimateGui") then
        PlayerGui.NDSUltimateGui:Destroy()
    end

    -- 1. Membuat ScreenGui Utama
    local ScreenGui = Instance.new("ScreenGui")
    ScreenGui.Name = "NDSUltimateGui"
    ScreenGui.ResetOnSpawn = false
    ScreenGui.Parent = PlayerGui

    -- Frame Utama (Menu - Ukuran Pendek dengan Scroll)
    local MainFrame = Instance.new("Frame")
    MainFrame.Size = UDim2.new(0, 260, 0, 310)
    MainFrame.Position = UDim2.new(0.1, 0, 0.2, 0)
    MainFrame.BackgroundColor3 = Color3.fromRGB(20, 20, 30)
    MainFrame.BorderSizePixel = 0
    MainFrame.Active = true
    MainFrame.Parent = ScreenGui

    local UICorner = Instance.new("UICorner")
    UICorner.CornerRadius = UDim.new(0, 12)
    UICorner.Parent = MainFrame

    local UIStroke = Instance.new("UIStroke")
    UIStroke.Color = Color3.fromRGB(0, 170, 255)
    UIStroke.Thickness = 2
    UIStroke.Parent = MainFrame

    -- Header untuk Dragging Manual
    local TopBar = Instance.new("Frame")
    TopBar.Size = UDim2.new(1, 0, 0, 40)
    TopBar.BackgroundTransparency = 1
    TopBar.Parent = MainFrame

    local Title = Instance.new("TextLabel")
    Title.Size = UDim2.new(1, 0, 1, 0)
    Title.BackgroundTransparency = 1
    Title.Text = "🌪️ NDS ULTIMATE GUI"
    Title.TextColor3 = Color3.fromRGB(255, 255, 255)
    Title.TextSize = 16
    Title.Font = Enum.Font.GothamBold
    Title.Parent = TopBar

    -- Skrip Dragging Manual
    local dragging, dragInput, dragStart, startPos
    TopBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = MainFrame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    TopBar.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)

    -- 2. ScrollingFrame
    local ScrollContainer = Instance.new("ScrollingFrame")
    ScrollContainer.Size = UDim2.new(1, -10, 1, -55)
    ScrollContainer.Position = UDim2.new(0, 5, 0, 45)
    ScrollContainer.BackgroundTransparency = 1
    ScrollContainer.BorderSizePixel = 0
    ScrollContainer.CanvasSize = UDim2.new(0, 0, 0, 435)
    ScrollContainer.ScrollBarThickness = 4
    ScrollContainer.Parent = MainFrame

    -- Fungsi membuat Tombol
    local function createButton(name, posY)
        local btn = Instance.new("TextButton")
        btn.Size = UDim2.new(0, 230, 0, 38)
        btn.Position = UDim2.new(0.5, -115, 0, posY)
        btn.BackgroundColor3 = Color3.fromRGB(35, 35, 50)
        btn.TextColor3 = Color3.fromRGB(255, 255, 255)
        btn.TextSize = 13
        btn.Font = Enum.Font.GothamMedium
        btn.Text = name
        btn.Parent = ScrollContainer

        local corner = Instance.new("UICorner")
        corner.CornerRadius = UDim.new(0, 8)
        corner.Parent = btn

        local stroke = Instance.new("UIStroke")
        stroke.Color = Color3.fromRGB(0, 170, 255)
        stroke.Transparency = 0.5
        stroke.Parent = btn

        return btn
    end

    -- Daftar Tombol Fitur
    local FlyBtn = createButton("Fly Mode: OFF", 5)
    local NoClipBtn = createButton("No Clip: OFF", 48)
    local WalkInWaterBtn = createButton("Walk in Water: OFF", 91)
    local AntiFlingBtn = createButton("Anti-Fling: ON", 134)
    AntiFlingBtn.BackgroundColor3 = Color3.fromRGB(0, 120, 80)

    local FlingBtn = createButton("Fling Player: OFF", 177)

    -- Speed Label & Input Box
    local SpeedLabel = Instance.new("TextLabel")
    SpeedLabel.Size = UDim2.new(0, 230, 0, 18)
    SpeedLabel.Position = UDim2.new(0.5, -115, 0, 220)
    SpeedLabel.BackgroundTransparency = 1
    SpeedLabel.Text = "Custom Speed (Default: 16):"
    SpeedLabel.TextColor3 = Color3.fromRGB(200, 200, 200)
    SpeedLabel.TextSize = 11
    SpeedLabel.Font = Enum.Font.Gotham
    SpeedLabel.Parent = ScrollContainer

    local SpeedBox = Instance.new("TextBox")
    SpeedBox.Size = UDim2.new(0, 230, 0, 30)
    SpeedBox.Position = UDim2.new(0.5, -115, 0, 241)
    SpeedBox.BackgroundColor3 = Color3.fromRGB(30, 30, 45)
    SpeedBox.TextColor3 = Color3.fromRGB(0, 255, 120)
    SpeedBox.TextSize = 13
    SpeedBox.Font = Enum.Font.GothamBold
    SpeedBox.Text = "50"
    SpeedBox.ClearTextOnFocus = false
    SpeedBox.Parent = ScrollContainer

    local BoxCorner = Instance.new("UICorner")
    BoxCorner.CornerRadius = UDim.new(0, 6)
    BoxCorner.Parent = SpeedBox

    local SpeedBtn = createButton("Speed Hack: OFF", 278)
    local InfiniteYieldBtn = createButton("Infinite Yield", 321)
    local CloseBtn = createButton("Minimize GUI (-)", 364)

    -- 3. TOMBOL LOGO TORNADO (Minimize / Open)
    local OpenButton = Instance.new("TextButton")
    OpenButton.Size = UDim2.new(0, 50, 0, 50)
    OpenButton.Position = UDim2.new(0.02, 0, 0.2, 0)
    OpenButton.BackgroundColor3 = Color3.fromRGB(20, 20, 30)
    OpenButton.TextColor3 = Color3.fromRGB(0, 170, 255)
    OpenButton.TextSize = 22
    OpenButton.Font = Enum.Font.GothamBold
    OpenButton.Text = "🌪️"
    OpenButton.Visible = false
    OpenButton.Active = true
    OpenButton.Parent = ScreenGui

    local OpenCorner = Instance.new("UICorner")
    OpenCorner.CornerRadius = UDim.new(1, 0)
    OpenCorner.Parent = OpenButton

    local OpenStroke = Instance.new("UIStroke")
    OpenStroke.Color = Color3.fromRGB(0, 170, 255)
    OpenStroke.Thickness = 2
    OpenStroke.Parent = OpenButton

    -- Dragging Logo Tornado
    local logoDragging, logoDragStart, logoStartPos
    OpenButton.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            logoDragging = true
            logoDragStart = input.Position
            logoStartPos = OpenButton.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if logoDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - logoDragStart
            OpenButton.Position = UDim2.new(logoStartPos.X.Scale, logoStartPos.X.Offset + delta.X, logoStartPos.Y.Scale, logoStartPos.Y.Offset + delta.Y)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            logoDragging = false
        end
    end)

    CloseBtn.MouseButton1Click:Connect(function()
        MainFrame.Visible = false
        OpenButton.Visible = true
    end)

    OpenButton.MouseButton1Click:Connect(function()
        MainFrame.Visible = true
        OpenButton.Visible = false
    end)

    -- ==================== LOGIC / FITUR SCRIPTS ====================

    -- A. WALK IN WATER
    local waterWalkActive = false
    local waterPart = nil

    WalkInWaterBtn.MouseButton1Click:Connect(function()
        waterWalkActive = not waterWalkActive
        if waterWalkActive then
            WalkInWaterBtn.Text = "Walk in Water: ON"
            WalkInWaterBtn.BackgroundColor3 = Color3.fromRGB(0, 120, 80)

            waterPart = Instance.new("Part")
            waterPart.Name = "WaterWalkPlatform"
            waterPart.Size = Vector3.new(15, 1, 15)
            waterPart.Transparency = 1
            waterPart.Anchored = true
            waterPart.CanCollide = true
            waterPart.Parent = Workspace

            task.spawn(function()
                while waterWalkActive do
                    local char = LocalPlayer.Character
                    local root = char and char:FindFirstChild("HumanoidRootPart")
                    if root then
                        waterPart.Position = Vector3.new(root.Position.X, 0, root.Position.Z)
                    end
                    RunService.RenderStepped:Wait()
                end
                if waterPart then
                    waterPart:Destroy()
                    waterPart = nil
                end
            end)
        else
            WalkInWaterBtn.Text = "Walk in Water: OFF"
            WalkInWaterBtn.BackgroundColor3 = Color3.fromRGB(35, 35, 50)
            if waterPart then
                waterPart:Destroy()
                waterPart = nil
            end
        end
    end)

    -- B. ANTI-FLING (FIXED & AMAN)
    local antiFlingActive = true
    AntiFlingBtn.MouseButton1Click:Connect(function()
        antiFlingActive = not antiFlingActive
        if antiFlingActive then
            AntiFlingBtn.Text = "Anti-Fling: ON"
            AntiFlingBtn.BackgroundColor3 = Color3.fromRGB(0, 120, 80)
        else
            AntiFlingBtn.Text = "Anti-Fling: OFF"
            AntiFlingBtn.BackgroundColor3 = Color3.fromRGB(35, 35, 50)
        end
    end)

    RunService.Stepped:Connect(function()
        if antiFlingActive and LocalPlayer.Character then
            local root = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
            if root then
                root.AssemblyLinearVelocity = Vector3.new(0, root.AssemblyLinearVelocity.Y, 0)
                root.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
            end
        end
    end)

    -- C. INFINITE YIELD
    InfiniteYieldBtn.MouseButton1Click:Connect(function()
        InfiniteYieldBtn.Text = "Loading IY..."
        pcall(function()
            loadstring(game:HttpGet('https://raw.githubusercontent.com/EdgeIY/infiniteyield/master/source'))()
        end)
        InfiniteYieldBtn.Text = "Infinite Yield (Loaded!)"
    end)

    -- D. SPEED HACK
    local speedEnabled = false
    SpeedBtn.MouseButton1Click:Connect(function()
        speedEnabled = not speedEnabled
        if speedEnabled then
            SpeedBtn.Text = "Speed Hack: ON"
            SpeedBtn.BackgroundColor3 = Color3.fromRGB(0, 120, 80)
        else
            SpeedBtn.Text = "Speed Hack: OFF"
            SpeedBtn.BackgroundColor3 = Color3.fromRGB(35, 35, 50)
            if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
                LocalPlayer.Character.Humanoid.WalkSpeed = 16
            end
        end
    end)

    RunService.RenderStepped:Connect(function()
        if speedEnabled and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
            local customSpeed = tonumber(SpeedBox.Text) or 16
            LocalPlayer.Character.Humanoid.WalkSpeed = customSpeed
        end
    end)

    -- E. FLY MODE
    local flying = false
    local bv, bg
    FlyBtn.MouseButton1Click:Connect(function()
        flying = not flying
        local char = LocalPlayer.Character
        if not char or not char:FindFirstChild("HumanoidRootPart") then return end

        if flying then
            FlyBtn.Text = "Fly Mode: ON"
            FlyBtn.BackgroundColor3 = Color3.fromRGB(0, 120, 80)

            local rootPart = char.HumanoidRootPart
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then hum.PlatformStand = true end

            bv = Instance.new("BodyVelocity")
            bv.Velocity = Vector3.new(0, 0, 0)
            bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
            bv.Parent = rootPart

            bg = Instance.new("BodyGyro")
            bg.CFrame = rootPart.CFrame
            bg.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
            bg.Parent = rootPart

            task.spawn(function()
                while flying and char and char:FindFirstChild("HumanoidRootPart") and hum do
                    local cam = Workspace.CurrentCamera
                    bg.CFrame = cam.CFrame

                    if hum.MoveDirection.Magnitude > 0 then
                        bv.Velocity = cam.CFrame.LookVector * (hum.MoveDirection.Magnitude * 50)
                    else
                        bv.Velocity = Vector3.new(0, 0.1, 0)
                    end
                    RunService.RenderStepped:Wait()
                end
            end)
        else
            FlyBtn.Text = "Fly Mode: OFF"
            FlyBtn.BackgroundColor3 = Color3.fromRGB(35, 35, 50)
            if char and char:FindFirstChildOfClass("Humanoid") then
                char.Humanoid.PlatformStand = false
            end
            if bv then bv:Destroy() end
            if bg then bg:Destroy() end
        end
    end)

    -- F. NO CLIP
    local noclipEnabled = false
    NoClipBtn.MouseButton1Click:Connect(function()
        noclipEnabled = not noclipEnabled
        if noclipEnabled then
            NoClipBtn.Text = "No Clip: ON"
            NoClipBtn.BackgroundColor3 = Color3.fromRGB(0, 120, 80)
        else
            NoClipBtn.Text = "No Clip: OFF"
            NoClipBtn.BackgroundColor3 = Color3.fromRGB(35, 35, 50)
        end
    end)

    RunService.Stepped:Connect(function()
        if noclipEnabled and LocalPlayer.Character then
            for _, part in pairs(LocalPlayer.Character:GetDescendants()) do
                if part:IsA("BasePart") then
                    part.CanCollide = false
                end
            end
        end
    end)

    -- G. FLING PLAYER
    local flingActive = false
    FlingBtn.MouseButton1Click:Connect(function()
        flingActive = not flingActive
        if flingActive then
            FlingBtn.Text = "Fling Player: ON"
            FlingBtn.BackgroundColor3 = Color3.fromRGB(180, 50, 50)

            noclipEnabled = true
            NoClipBtn.Text = "No Clip: ON"
            NoClipBtn.BackgroundColor3 = Color3.fromRGB(0, 120, 80)

            task.spawn(function()
                local character = LocalPlayer.Character
                local root = character and character:FindFirstChild("HumanoidRootPart")
                if not root then return end

                local bav = Instance.new("BodyAngularVelocity")
                bav.Name = "FlingSpin"
                bav.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
                bav.AngularVelocity = Vector3.new(0, 99999, 0)
                bav.Parent = root

                while flingActive do
                    RunService.RenderStepped:Wait()
                end
                if bav then bav:Destroy() end
            end)
        else
            FlingBtn.Text = "Fling Player: OFF"
            FlingBtn.BackgroundColor3 = Color3.fromRGB(35, 35, 50)
            local character = LocalPlayer.Character
            if character and character:FindFirstChild("HumanoidRootPart") and character.HumanoidRootPart:FindFirstChild("FlingSpin") then
                character.HumanoidRootPart.FlingSpin:Destroy()
            end
        end
    end)

end
